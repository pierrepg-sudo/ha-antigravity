# Adapted from github.com/moby/profiles/apparmor/template.go.
# Modifications: standalone add-on name and peer labels; root propagation, two exact tmpfs mounts and one exact proc mount.
# Experimental profile: verified root propagation, shm, staging-root and proc mounts.
# Runtime isolation is NOT verified. CLI command approval must remain enabled.
abi <abi/3.0>,
#include <tunables/global>

# Apply the same proc protections before and after the sandbox root transition.
@{AGY_PROC}=@{PROC} /dev/shm/setup/root/proc/

profile antigravity_remote flags=(attach_disconnected,mediate_deleted) {
  # Compatibility baseline modeled on Docker's default container profile.
  # Reconcile against the target Docker version before any test deployment.
  # These AppArmor permissions do not grant Linux capabilities or bypass DAC.
  #include <abstractions/base>
  network,
  capability,
  file,
  umount,

  # Preserve the upstream protocol restrictions represented in the reviewed template.
  deny network ax25,
  deny network ipx,
  deny network appletalk,
  deny network netrom,
  deny network rose,
  deny network econet,
  deny network atmsvc,
  deny network irda,
  deny network wanpipe,
  deny network isdn,
  deny network caif,
  deny network alg,
  deny network vsock,

  # Host runtime may stop the container; peers remain within this profile.
  # Supervisor renames only the declaration; the built-in variable follows it.
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,
  signal (receive) peer=crun,
  signal (receive) peer=hassio-supervisor,
  signal (send,receive) peer=@{profile_name},
  ptrace (trace,tracedby,read,readby) peer=@{profile_name},

  deny @{AGY_PROC}/* w,
  deny @{AGY_PROC}/{[^1-9/],[^1-9/][^0-9/],[^1-9s/][^0-9y/][^0-9s/],[^1-9/][^0-9/][^0-9/][^0-9/]*}/** w,
  deny @{AGY_PROC}/sys/[^k]** w,
  deny @{AGY_PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{AGY_PROC}/sysrq-trigger rwklx,
  deny @{AGY_PROC}/kcore rwklx,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,

  # Root propagation exceptions required by native startup and the namespace probe.
  # Exact option sets: no "in (...)" wildcard combinations.
  # Log text does not distinguish MS_PRIVATE from MS_PRIVATE|MS_REC;
  # retain only the variant(s) shown necessary by native/probe audit evidence.
  audit mount options=(private) -> /,
  audit mount options=(rprivate) -> /,

  # CLI 1.3.1 ARM64 jailMain: syscall.Mount("", "/dev/shm", "tmpfs", 0, "").
  # options=(rw) matches zero mount flags; no bind/remount/move variants.
  # No device source is involved for tmpfs. See design/apparmor/CLI-1.3.1.md.
  audit mount fstype=tmpfs options=(rw) -> /dev/shm/,
  # CLI 1.3.1 jailMain at 0x69b06e8: empty source/data, flags=0.
  # Literal staging root only: no wildcard for its descendants.
  audit mount fstype=tmpfs options=(rw) -> /dev/shm/setup/root/,

  # CLI 1.3.1 jailMain at 0x69b0a04: type proc, flags 0xe.
  audit mount fstype=proc options=(rw,nosuid,nodev,noexec) -> /dev/shm/setup/root/proc/,

  # All other mount operations are implicitly denied.
  # Do NOT add "deny mount," here: explicit deny would override the exceptions.
  # No bind/rbind, other tmpfs/proc targets, remount, move or pivot_root grants.
}
