# Adapted from github.com/moby/profiles/apparmor/template.go.
# Modifications: standalone add-on name and peer labels; separate restricted command worker.
# Command worker must pass live isolation checks before accepting requests.
abi <abi/3.0>,
#include <tunables/global>

# Apply the same proc protections before and after the sandbox root transition.
@{AGY_PROC}=@{PROC}

profile antigravity_remote flags=(attach_disconnected,mediate_deleted) {
  # Compatibility baseline modeled on Docker's default container profile.
  # Reconcile against the target Docker version before any test deployment.
  # These AppArmor permissions do not grant Linux capabilities or bypass DAC.
  #include <abstractions/base>
  network,
  capability,
  file,
  umount,

  # Preserve the upstream protocol restrictions represented in the reviewed template.
  deny network ax25,
  deny network ipx,
  deny network appletalk,
  deny network netrom,
  deny network rose,
  deny network econet,
  deny network atmsvc,
  deny network irda,
  deny network wanpipe,
  deny network isdn,
  deny network caif,
  deny network alg,
  deny network vsock,

  # Host runtime may stop the container; peers remain within this profile.
  # Supervisor renames only the declaration; the built-in variable follows it.
  signal (receive) peer=unconfined,
  signal (receive) peer=runc,
  signal (receive) peer=crun,
  signal (receive) peer=hassio-supervisor,
  signal (send,receive) peer=@{profile_name},
  ptrace (trace,tracedby,read,readby) peer=@{profile_name},

  deny @{AGY_PROC}/* w,
  deny @{AGY_PROC}/{[^1-9/],[^1-9/][^0-9/],[^1-9s/][^0-9y/][^0-9s/],[^1-9/][^0-9/][^0-9/][^0-9/]*}/** w,
  deny @{AGY_PROC}/sys/[^k]** w,
  deny @{AGY_PROC}/sys/kernel/{?,??,[^s][^h][^m]**} w,
  deny @{AGY_PROC}/sysrq-trigger rwklx,
  deny @{AGY_PROC}/kcore rwklx,
  deny /sys/[^f]*/** wklx,
  deny /sys/f[^s]*/** wklx,
  deny /sys/fs/[^c]*/** wklx,
  deny /sys/fs/c[^g]*/** wklx,
  deny /sys/firmware/** rwklx,
  deny /sys/devices/virtual/powercap/** rwklx,
  deny /sys/kernel/security/** rwklx,

  # The broker can enter only the restricted child; no native sandbox mounts.
  change_profile -> @{profile_name}//command_worker,
  signal (send) peer=@{profile_name}//command_worker,

  profile command_worker flags=(attach_disconnected,mediate_deleted) {
    # No generic file, capability, network, mount or ptrace grants.
    /usr/** rixm,
    /bin/** rixm,
    /lib/** rixm,
    /lib64/** rixm,
    /etc/ld.so.cache r,
    /etc/{passwd,group,nsswitch.conf,localtime} r,
    /etc/fonts/** r,
    /etc/texmf/** r,
    /var/lib/texmf/** r,
    /var/cache/fontconfig/** r,
    /dev/{null,zero,random,urandom} rw,
    /proc/@{pid}/attr/current r,
    /proc/@{pid}/status r,
    /proc/meminfo r,
    /proc/cpuinfo r,
    /data/inputs/ r,
    /data/inputs/** r,
    /data/workspace/outputs/ rw,
    /data/workspace/outputs/** rwixm,
    /tmp/agy-worker/ r,
    /tmp/agy-worker/job-*/ rw,
    /tmp/agy-worker/job-*/** rwixm,
    signal (receive),
  }
}
